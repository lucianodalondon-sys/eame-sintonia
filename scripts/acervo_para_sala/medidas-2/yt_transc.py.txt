import json, os, re, glob, collections
M = json.load(open(r'C:/Users/London1/acervo-para-sala/motivos.json', encoding='utf-8'))
itens = M['ITENS'] if isinstance(M, dict) and 'ITENS' in M else M
raws = json.load(open(r'C:/Users/London1/acervo-para-sala/raw.json', encoding='utf-8'))
url = {r['sha256'].strip(): r.get('source_url') or '' for r in raws}
yt = [x for x in itens if 'YOUTUBE' in x.get('MARCAS', []) and x['MOTIVO'].startswith('NUNCA')]
ids = {}
for x in yt:
    m = re.search(r'v=([A-Za-z0-9_-]{11})', url.get(x['SHA256'], '') or x.get('SOURCE_URL', '') or '')
    if m:
        ids[m.group(1)] = x['SHA256']
print('youtube nunca perguntados', len(yt), 'com id de video', len(ids))
W = r'C:/Users/London1/orca/workspaces/eame-sintonia'
pads = [W + '/*/data/samples/YOUTUBE-TRANSCRICOES/**/*', W + '/yt-caption-integra/**/*',
        W + '/*/NAO_SEI/derivados/TRANSCRIPTION/*', W + '/*/data/samples/SCRAP-YOUTUBE/*',
        W + '/*/data/samples/YOUTUBE-RELEVANCIA/**/*', r'C:/Users/London1/sintonia-sala-italia/armazem/**/TRANSCRIPTION/*',
        W + '/*/data/samples/YOUTUBE*/**/*']
achados = collections.defaultdict(set)
vistos = set()
for p in pads:
    for f in glob.glob(p, recursive=True):
        if not os.path.isfile(f) or os.path.getsize(f) > 60_000_000:
            continue
        rp = os.path.realpath(f)
        k = (os.path.basename(f), os.path.getsize(f))
        if k in vistos:
            continue
        vistos.add(k)
        try:
            t = open(f, encoding='utf-8', errors='replace').read()
        except OSError:
            continue
        for v in set(re.findall(r'[A-Za-z0-9_-]{11}', t)) & ids.keys():
            achados[v].add(f.replace(W, '~')[:110])
print('ficheiros lidos', len(vistos))
print('videos com alguma mencao', len(achados))
c = collections.Counter()
for v, fs in achados.items():
    for f in fs:
        c[re.sub(r'[^/\\]*$', '', f.split('~', 1)[-1])[:80]] += 1
for k, n in c.most_common(20):
    print(n, k)
