AQUI = os.path.dirname(os.path.abspath(__file__))
A = {r["DOMINIO"]: r for r in json.load(open(os.path.join(AQUI, "ANCORAS.json"), encoding="utf-8"))}
HOJE = "2026-09-26"


def prova(d):
    r = A[d]
    return ("PROVA_DE_IDENTIDADE=pagina oficial aberta em %s pelo portao da casa (VPN IT, robots lido) · HTTP %s · "
            "sha256=%s · titulo «%s»%s%s" % (
                HOJE, r["STATUS"], r["SHA256"][:16], r["TITULO"][:90],
                (" · P.IVA/CF %s" % ",".join(r["PIVA"])) if r.get("PIVA") else "",
                (" · sede «%s»" % r["MORADA"][0]) if r.get("MORADA") else ""))


def ancora(d, pal):
    for u, t in A[d]["ANCORAS"]:
        if re.search(pal, t + " " + u, re.I):
            return u, t
    raise SystemExit("sem ancora %s em %s" % (pal, d))
