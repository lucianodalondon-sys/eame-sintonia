def main():
    env = dict(os.environ, PYTHONUTF8="1", HTTP_PROXY="http://127.0.0.1:9", HTTPS_PROXY="http://127.0.0.1:9")
    mortos = 0
    for nome, f, a, b in M:
        p = os.path.join(RAIZ, f)
        orig = io.open(p, encoding="utf-8", newline="").read()
        nl = "\r\n" if "\r\n" in orig else "\n"
        s = orig.replace("\r\n", "\n")
        if s.count(a) != 1:
            print(nome, "NAO_APLICOU", s.count(a))
            continue
        try:
            io.open(p, "w", encoding="utf-8", newline="").write(s.replace(a, b).replace("\n", nl))
            d = subprocess.run(["git", "diff", "--quiet", "--", f], cwd=RAIZ).returncode
            r = subprocess.run([sys.executable, "-m", "unittest", *TESTES], cwd=RAIZ, capture_output=True,
                               text=True, env=env, timeout=600)
            morto = r.returncode != 0
            mortos += morto
            print(nome, "MORTO" if morto else "SOBREVIVEU", "(diff ok)" if d == 1 else "(SEM DIFF!)")
        finally:
            io.open(p, "w", encoding="utf-8", newline="").write(orig)
    print("MUTACAO %d/%d" % (mortos, len(M)))


if __name__ == "__main__":
    main()
