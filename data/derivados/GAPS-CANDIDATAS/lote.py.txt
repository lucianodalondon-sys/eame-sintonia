def o_que_falta(motivo):
    if "SEM_FAMILIA" in motivo:
        return ("a FAMILIA de itens: a entrada nao lista 2+ boletins com o mesmo esqueleto; provar onde a lista "
                "dos boletins esta (outra pagina, PDF, arquivo por ano)")
    if "FAMILIA_ESTATICA" in motivo or "SERVICO_OU_INSTITUCIONAL" in motivo:
        return "que existe uma lista DATADA de boletins/avisos (a entrada so tem paginas fixas de servico)"
    if "ITEM_NAO_E_MATERIA" in motivo:
        u = re.search(r"https?://\S+", motivo)
        return ("que o item aberto e um BOLETIM (cultura + zona + data); o canario abriu "
                + (u.group(0)[:90] if u else "outra coisa"))
    if "ENTRADA_NAO_E_HTML" in motivo:
        return ("a FORMA: a entrada nao e HTML (PDF ou outro); provar a forma A da D42 (boletim em PDF) e a lista "
                "onde os PDF se publicam")
    if "ALVO_ERRADO" in motivo:
        return "o ALVO certo: o padrao do contrato aponta para outra seccao (revisao R1)"
    if "ENTRADA_E_MATERIA" in motivo:
        return "a pagina-indice: a entrada do contrato e um so boletim"
    if "DUPLICADA" in motivo:
        return "a familia do LISTINO (datado, por produto), distinta das noticias ja colhidas"
    return "tudo: QUALIFY (territorio), contrato e canario"
