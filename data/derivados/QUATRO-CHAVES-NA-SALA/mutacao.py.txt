    python3 -m unittest tests.test_quatro_chaves tests.test_quatro_chaves_na_sala
contra um Postgres DESCARTAVEL (o do proprio teste). Morto = os testes ficaram vermelhos.
Antes dos mutantes corre o CONTROLO (sem defeito), que tem de dar verde.

    uso: python3 mutacao.py.txt <raiz-do-repo> [saida.txt]
    CORRER_COMO=postgres  -> corre os testes com `runuser -u postgres` (initdb recusa root)
"""

def correr(arvore):
    cmd = [sys.executable, "-m", "unittest", "tests.test_quatro_chaves", "tests.test_quatro_chaves_na_sala"]
    if COMO:
        cmd = ["runuser", "-u", COMO, "--", "env", "PYTHONDONTWRITEBYTECODE=1", "HOME=/tmp"] + cmd
    r = subprocess.run(cmd, cwd=arvore, capture_output=True, text=True, timeout=900)
    ult = [l for l in r.stderr.splitlines() if l.strip()][-1:] or ["?"]
    corridos = [l for l in r.stderr.splitlines() if l.startswith("Ran ")]
    return r.returncode, "%s · %s" % (corridos[-1] if corridos else "?", ult[0])


def copia(base):
    d = tempfile.mkdtemp(prefix="qc-mut-")
    shutil.rmtree(d); shutil.copytree(base, d)
    os.chmod(d, 0o777)
    for r, ds, fs in os.walk(d):
        for x in ds + fs:
            os.chmod(os.path.join(r, x), 0o777)
    return d


def main():
    base = tempfile.mkdtemp(prefix="qc-base-")
    subprocess.run("git -C '%s' archive HEAD | tar -x -C '%s'" % (RAIZ, base), shell=True, check=True)
    linhas = []
    d = copia(base)
    rc, res = correr(d); shutil.rmtree(d, ignore_errors=True)
    linhas.append("CONTROLO (sem defeito): rc=%d  %s  -> %s" % (rc, res, "VERDE" if rc == 0 else "VERMELHO (!!)"))
    mortos = 0
    for nome, f, antes, depois in MUTANTES:
        d = copia(base)
        p = os.path.join(d, f)
        s = open(p, encoding="utf-8").read()
        n = s.count(antes)
        if n != 1:
            linhas.append("%s: NAO APLICADO (trecho encontrado %d vezes)" % (nome, n)); continue
        open(p, "w", encoding="utf-8").write(s.replace(antes, depois))
        rc, res = correr(d); shutil.rmtree(d, ignore_errors=True)
        morto = rc != 0
        mortos += morto
        linhas.append("%s: %s  (rc=%d; %s)" % (nome, "MORTO" if morto else "SOBREVIVEU", rc, res))
    linhas.append("MORTOS %d / %d" % (mortos, len(MUTANTES)))
    shutil.rmtree(base, ignore_errors=True)
    txt = "\n".join(linhas) + "\n"
    print(txt)
    if SAIDA:
        open(SAIDA, "w", encoding="utf-8").write(txt)


main()
