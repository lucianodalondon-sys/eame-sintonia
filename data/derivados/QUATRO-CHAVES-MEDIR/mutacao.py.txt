# QUATRO-CHAVES-MEDIR · mutacao de listar_pendentes POR NOME. Registo do ensaio (.txt).
#   PYTHONDONTWRITEBYTECODE=1 PYTHONUTF8=1 py data/derivados/QUATRO-CHAVES-MEDIR/mutacao.py.txt
import hashlib
import os
import subprocess
import sys

RAIZ = os.path.dirname(os.path.dirname(os.path.dirname(os.path.dirname(
    os.path.abspath(__file__)))))
F = "admissao/sala_de_espera.py"
T = "tests.test_sala_por_nome"

MUTANTES = [
    ("M6 tabela: a ordem e o item trocam de chave",
     '("run_id", "RUN_ID"), ("ordem", "ORDEM"), ("item_id", "ITEM_ID"),',
     '("run_id", "RUN_ID"), ("ordem", "ITEM_ID"), ("item_id", "ORDEM"),'),
    ("M7 leitura: sem conferir o numero de colunas",
     '            if len(partes) != len(COLUNAS_PENDENTES):',
     '            if False:'),
    ("M8 leitura: volta a ler por posicao (o codigo antigo)",
     '            u = {campo: v for (_, campo), v in zip(COLUNAS_PENDENTES, partes)}',
     '            u = {"RUN_ID": partes[0], "ORDEM": partes[1], "ITEM_ID": partes[2]}'),
]


def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


mortos = 0
caminho = os.path.join(RAIZ, F)
for nome, de, para in MUTANTES:
    with open(caminho, "rb") as f:
        original = f.read()
    antes = sha(caminho)
    texto = original.decode("utf-8").replace("\r\n", "\n")
    assert texto.count(de) == 1, "%s: o alvo da troca nao e unico (%d)" % (nome, texto.count(de))
    try:
        with open(caminho, "wb") as f:
            f.write(texto.replace(de, para).encode("utf-8"))
        r = subprocess.run([sys.executable, "-m", "unittest", T], cwd=RAIZ, capture_output=True,
                           text=True, env=dict(os.environ, PYTHONDONTWRITEBYTECODE="1", PYTHONUTF8="1"))
        morto = r.returncode != 0
    finally:
        with open(caminho, "wb") as f:
            f.write(original)
    reposto = sha(caminho) == antes
    mortos += morto
    print("%-52s %s · reposto=%s" % (nome, "MORTO (o teste apanhou)" if morto
                                     else "SOBREVIVEU  <-- o teste nao morde",
                                     "SIM" if reposto else "NAO <-- GRAVE"))
    if not reposto:
        sys.exit(2)
print("MUTANTES %d · MORTOS %d" % (len(MUTANTES), mortos))
sys.exit(0 if mortos == len(MUTANTES) else 1)
