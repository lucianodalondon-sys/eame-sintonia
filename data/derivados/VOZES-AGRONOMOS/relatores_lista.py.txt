def video(sid, trecho):
    for v in T.get(sid, {}).get("TITULOS", []):
        if trecho.lower() in v["TITULO"].lower():
            return {"SOURCE_ID": sid, "TITULO": v["TITULO"], "URL": v["URL"], "SHA256_BRUTO": v["SHA256"]}
    raise SystemExit("video nao achado: %s %s" % (sid, trecho))
