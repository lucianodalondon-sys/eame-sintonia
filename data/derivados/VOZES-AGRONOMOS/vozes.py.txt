        assert video, (vid, trecho)
    bloqueado = next((p for p in PROIBIDOS if p in dom), None)

# plano de pedidos: por dominio, <=5 por ronda (D38). Cada pessoa: 1 pedido a pagina institucional (prova do papel).
# Videos: pela API oficial (googleapis.com, 1 videos.list por lote de 50 ids = descricao + data + canal), nao /watch.
por_dom = collections.defaultdict(list)
descobrir = collections.defaultdict(list)
for f in fichas:
    if f["BLOQUEIO_DA_COORDENACAO"]:
        continue
    if f["PROVAR_PAPEL_EM"].startswith("DESCOBRIR"):
        descobrir[f["PROVAR_PAPEL_EM"]].append(f["ID"])     # nao se inventa endereco: descoberta primeiro
    else:
        por_dom[f["PROVAR_PAPEL_EM"]].append(f["ID"])
plano = []
for dom, ids in sorted(por_dom.items(), key=lambda kv: (-len(kv[1]), kv[0])):
    rondas = [ids[i:i + 5] for i in range(0, len(ids), 5)]
    plano.append({"DOMINIO": dom, "PEDIDOS": len(ids), "RONDAS": len(rondas), "IDS_POR_RONDA": rondas})
ids_video = [f["VIDEO_GUARDADO"]["URL"] for f in fichas if f["VIDEO_GUARDADO"]]
plano.append({"DOMINIO": "www.googleapis.com (YouTube Data API, via Scrap/D53)", "PEDIDOS": 1, "RONDAS": 1,
              "IDS_POR_RONDA": [["videos.list com %d ids (descricao, data, canal)" % len(ids_video)]],
              "NOTA": "a chave so existe no segredo do GitHub (runbook CANAIS-41); 0 pedidos a youtube.com/watch"})
