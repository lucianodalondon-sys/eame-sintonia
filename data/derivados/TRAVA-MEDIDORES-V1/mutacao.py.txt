# -*- coding: utf-8 -*-
# Guardado como .txt de proposito: e o registo do ensaio de mutacao da missao
# TRAVA-MEDIDORES-V1, nao codigo do sistema. Corre-se com:
#     PYTHONDONTWRITEBYTECODE=1 py data/derivados/TRAVA-MEDIDORES-V1/mutacao.py.txt
# Cada mutante: aplica UMA troca, corre SO a classe de teste que o deve apanhar,
# repoe os bytes originais e confere o sha antes de passar ao seguinte.
import hashlib
import os
import subprocess
import sys

RAIZ = os.path.dirname(os.path.dirname(os.path.dirname(os.path.dirname(
    os.path.abspath(__file__)))))
E = "system-map/scripts/censo_das_estradas_it.py"
C = "system-map/scripts/censo_do_congelamento.py"
T = "tests.test_os_medidores_da_trava_veem."
VIGIA = T + "OVigiaDoCongelamentoComparaComAFotografia"
ESTRADAS = T + "OMedidorDasEstradasVeOLivroDoColetor"

MUTANTES = [
    ("M1 o vigia compara o blob consigo", C,
     'elif agora != a["GIT_BLOB_SHA"]:', 'elif agora != agora:', VIGIA),
    ("M2 FROZEN_AT_HEAD volta a ser o HEAD", C,
     '"FROZEN_AT_HEAD": manifesto["FROZEN_AT_HEAD"],\n        "O_QUE',
     '"FROZEN_AT_HEAD": cabeca,\n        "O_QUE', VIGIA),
    ("M3 o vigia ignora os novos", C,
     'for a in congelados_hoje if a["PATH"] not in na_foto]',
     'for a in congelados_hoje if False]', VIGIA),
    ("M4 o censo volta a ler so o catalogo", E,
     "with open(LIVRO_DO_COLETOR, encoding='utf-8') as f:\n        linhas",
     "with open(CATALOGO, encoding='utf-8') as f:\n        linhas", ESTRADAS),
    ("M5 o censo inventa classe de estrada", E,
     "'ROUTE_CLASS_ID': SEM_CLASSE,", "'ROUTE_CLASS_ID': 'RC-1',", ESTRADAS),
    ("M6 a sobreposicao conta tudo", E,
     "'TAMBEM_NO_CATALOGO_DA_TRAVA': len(ids & catalogo),",
     "'TAMBEM_NO_CATALOGO_DA_TRAVA': len(ids),", ESTRADAS),
]


def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


mortos = 0
for nome, rel, de, para, alvo in MUTANTES:
    caminho = os.path.join(RAIZ, rel)
    with open(caminho, "rb") as f:
        original = f.read()
    antes = sha(caminho)
    texto = original.decode("utf-8")
    texto_norm = texto.replace("\r\n", "\n")
    assert texto_norm.count(de) == 1, "%s: o alvo da troca nao e unico" % nome
    mutado = texto_norm.replace(de, para)
    try:
        with open(caminho, "wb") as f:
            f.write(mutado.encode("utf-8"))
        r = subprocess.run([sys.executable, "-m", "unittest", alvo], cwd=RAIZ,
                           capture_output=True, text=True,
                           env=dict(os.environ, PYTHONDONTWRITEBYTECODE="1",
                                    PYTHONUTF8="1"))
        morto = r.returncode != 0
    finally:
        with open(caminho, "wb") as f:
            f.write(original)
    reposto = sha(caminho) == antes
    mortos += morto
    print("%-42s %s · reposto=%s" % (nome, "MORTO (o teste apanhou)" if morto
                                     else "SOBREVIVEU  <-- o teste nao morde",
                                     "SIM" if reposto else "NAO <-- GRAVE"))
    if not reposto:
        sys.exit(2)
print("MUTANTES %d · MORTOS %d" % (len(MUTANTES), mortos))
sys.exit(0 if mortos == len(MUTANTES) else 1)
