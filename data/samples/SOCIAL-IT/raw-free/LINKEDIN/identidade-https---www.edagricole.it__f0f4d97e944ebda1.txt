</div><h2 style="color: #005a2c;text-align: center" class="vc_custom_heading vc_do_custom_heading">L'agricoltura firmata Edagricole, dal 1937</h2><div class="td_block_wrap td_block_text_with_title td_uid_2_6aaf7f2aead79_rand td-pb-border-top td_block_template_3" data-td-block-uid="td_uid_2_6aaf7f2aead79"><div class="td-block-title-wrap"></div><div class="td_mod_wrap td-fix-index"><p style="text-align: center;">Il marchio Edagricole, nato nel 1937 per contraddistinguere la produzione della prima casa editrice italiana interamente dedicata al settore agricolo, &egrave; oggi marchio leader nell&rsquo;informazione del settore agroalimentare, di propriet&agrave; della societ&agrave; <strong>Tecniche Nuove di Milano</strong>.</p>
<p style="text-align: center;">Alle riviste, diffuse a pagamento sia in abbonamento postale sia in versione digitale, si affianca un catalogo libri di oltre 300 titoli, nel quale si possono trovare sia volumi che approfondiscono le tematiche agronomiche, veterinarie, agroalimentari e paesaggistiche dal punto di vista tecnico e legislativo per un target di studiosi ed universitari, sia libri di taglio tecnico-pratico rivolti a chiunque operi, a qualsiasi livello, anche hobbistico, nell&rsquo;ambito dell&rsquo;agricoltura e di tematiche affini (giardinaggio, food).</p>
<p style="text-align: center;">Ogni rivista &egrave; anche un sito web costantemente aggiornato con news ed approfondimenti verticali e una porta di accesso ai servizi messi a disposizione gratuitamente per gli utenti abbonati.</p>
<p style="text-align: center;">Da anni Edagricole &egrave; inoltre fortemente impegnata nell&rsquo;ambito della formazione agli imprenditori agricoli e ai tecnici mediante il progetto Nova Agricoltura, che valorizza l&rsquo;innovazione sostenibile in agricoltura in una intensa attivit&agrave; di convegnistica in tutti i settori dell&rsquo;agricoltura moderna.</p>

                */
                modal.classList.add('cmplz-manage-consent', 'img-cmplz-tcf-popup-click-allowed');

                modal.setAttribute('role', 'dialog');
                modal.setAttribute('aria-modal', 'true');
                modal.setAttribute('aria-labelledby', config.modalId + '-title');
                modal.setAttribute('aria-hidden', 'true');



                dialog = createElement('div', 'img-cmplz-tcf-popup__dialog');
                dialog.classList.add('cmplz-manage-consent', 'img-cmplz-tcf-popup-click-allowed');
                dialog.setAttribute('role', 'document');




                header = createElement('div', 'img-cmplz-tcf-popup__header');

                titleWrap = createElement('div');

                title = createElement('h2', 'img-cmplz-tcf-popup__title', config.title);
                title.id = config.modalId + '-title';

                description = createElement('p', 'img-cmplz-tcf-popup__description', config.description);

                closeButton = createElement('button', 'img-cmplz-tcf-popup__close', '×');
                closeButton.type = 'button';
                closeButton.setAttribute('aria-label', config.closeLabel);

                body = createElement('div', 'img-cmplz-tcf-popup__body');

                iframe = createElement('iframe', 'img-cmplz-tcf-popup__iframe');
                iframe.setAttribute('title', config.title);
                iframe.setAttribute('loading', 'eager');
                iframe.setAttribute('referrerpolicy', 'same-origin');

                loading = createElement('div', 'img-cmplz-tcf-popup__loading', config.loadingLabel);
                saving = createElement('div', 'img-cmplz-tcf-popup__saving', config.savingLabel);
                errorBox = createElement('div', 'img-cmplz-tcf-popup__error is-hidden', config.missingUrlLabel);

                titleWrap.appendChild(title);
                titleWrap.appendChild(description);

                header.appendChild(titleWrap);
                header.appendChild(closeButton);

                body.appendChild(iframe);
                body.appendChild(loading);
                body.appendChild(saving);
                body.appendChild(errorBox);

                dialog.appendChild(header);
                dialog.appendChild(body);

                modal.appendChild(dialog);
                document.body.appendChild(modal);

                closeButton.addEventListener('click', requestCloseModal);

                modal.addEventListener('click', function (event) {
                    if (event.target === modal) {
                        requestCloseModal();
                    }
                });

                dialog.addEventListener('click', function (event) {
                    event.stopPropagation();
                });

                iframe.addEventListener('load', function () {
                    iframeIsLoading = false;
                    hideLoading();
                    hideSaving();
                    enhanceIframePage();
                });

                return modal;
            }

            function showLoading() {
                if (loading) {
                    loading.classList.remove('is-hidden');
                }

                if (errorBox) {
                    errorBox.classList.add('is-hidden');
                }
            }

            function hideLoading() {
                if (loading) {
                    loading.classList.add('is-hidden');
                }
            }

            function showSaving() {
                if (saving) {
                    saving.classList.add('is-visible');
                }
            }

            function hideSaving() {
                if (saving) {
                    saving.classList.remove('is-visible');
                }
            }

            function markConsentInteraction() {
                hasConsentInteraction = true;
                lastConsentInteractionAt = Date.now();
            }

            function hasRecentConsentInteraction() {
                if (!lastConsentInteractionAt) {
                    return false;
                }

                return (Date.now() - lastConsentInteractionAt) <= Number(config.recentInteractionWindowMs || 4000);
            }

            function hasPendingConsentWork() {
                return Boolean(
                    iframeIsLoading ||
                    pendingAjaxRequests > 0 ||
                    hasRecentConsentInteraction()
                );
            }

            function showError(message) {
                hideLoading();

                if (!errorBox) {
                    return;
                }

                errorBox.textContent = message || config.missingUrlLabel;
                errorBox.classList.remove('is-hidden');
            }

            function openModal(trigger) {
                var rawUrl;
                var popupUrl;

                createModal();

                previousActiveElement = trigger || document.activeElement;

                rawUrl = getTriggerUrl(trigger);

                modal.classList.add(config.visibleClass);
                modal.setAttribute('aria-hidden', 'false');

                document.documentElement.classList.add(config.openClass);

                if (document.body) {
                    document.body.classList.add(config.openClass);
                }

                window.setTimeout(function () {
                    var closeButton = modal.querySelector('.img-cmplz-tcf-popup__close');

                    if (closeButton && typeof closeButton.focus === 'function') {
                        closeButton.focus();
                    }
                }, 50);

                if (!rawUrl) {
                    if (iframe) {
                        iframe.removeAttribute('src');
                    }

                    showError(config.missingUrlLabel);
                    return;
                }

                popupUrl = normalizeCookiePolicyUrl(rawUrl);

                showLoading();

                if (iframe) {
                    iframeIsLoading = true;
                    pendingAjaxRequests = 0;
                    hasConsentInteraction = false;
                    lastConsentInteractionAt = 0;
                    closeInProgress = false;

                    iframe.removeAttribute('src');

                    window.setTimeout(function () {
                        iframe.src = popupUrl;
                    }, 10);
                }
            }

            function requestCloseModal() {
                var startedAt;

                if (!modal) {
                    return;
                }

                /*
                * Se non ci sono state interazioni recenti, chiudiamo subito.
                */
                if (!hasConsentInteraction && !hasPendingConsentWork()) {
                    finalizeCloseModal(false);
                    return;
                }

                /*
                * Evita doppi click sulla X.
                */
                if (closeInProgress) {
                    return;
                }

                closeInProgress = true;
                startedAt = Date.now();

                showSaving();

                function maybeFinalizeClose() {
                    var elapsed = Date.now() - startedAt;
                    var minDelayReached = elapsed >= Number(config.closeMinDelayMs || 1800);
                    var maxDelayReached = elapsed >= Number(config.closeMaxDelayMs || 5000);

                    /*
                    * Chiudiamo quando:
                    * - è passato il delay minimo;
                    * - non ci sono reload/ajax in corso;
                    *
                    * Oppure forziamo la chiusura al max timeout.
                    */
                    if (
                        (minDelayReached && !iframeIsLoading && pendingAjaxRequests <= 0) ||
                        maxDelayReached
                    ) {
                        finalizeCloseModal(hasConsentInteraction);
                        return;
                    }

                    window.setTimeout(maybeFinalizeClose, 150);
                }

                maybeFinalizeClose();
            }

            function finalizeCloseModal(shouldReloadParent) {
                if (!modal) {
                    return;
                }

                modal.classList.remove(config.visibleClass);
                modal.setAttribute('aria-hidden', 'true');

                document.documentElement.classList.remove(config.openClass);

                if (document.body) {
                    document.body.classList.remove(config.openClass);
                }

                hideLoading();
                hideSaving();

                /*
                * Rimuoviamo src solo dopo aver atteso eventuali salvataggi/reload.
                */
                if (iframe) {
                    iframe.removeAttribute('src');
                }

                if (
                    previousActiveElement &&
                    typeof previousActiveElement.focus === 'function' &&
                    document.documentElement.contains(previousActiveElement)
                ) {
                    previousActiveElement.focus();
                }

                previousActiveElement = null;
                iframeIsLoading = false;
                pendingAjaxRequests = 0;
                closeInProgress = false;

                /*
                * Se l'utente ha modificato qualcosa, ricarichiamo la pagina padre
                * per sincronizzare banner, cookie, TCF string e script sbloccati/bloccati.
                */
                if (shouldReloadParent && config.reloadParentOnConsent) {
                    window.setTimeout(function () {
                        try {
                            window.location.reload();
                        } catch (error) {
                            // noop.
                        }
                    }, 100);
                }

                hasConsentInteraction = false;
                lastConsentInteractionAt = 0;
            }

            function reloadParentAfterConsent() {
                if (!config.reloadParentOnConsent) {
                    closeModal();
                    return;
                }

                window.setTimeout(function () {
                    try {
                        window.location.reload();
                    } catch (error) {
                        closeModal();
                    }
                }, 1000);
            }

            function enhanceIframePage() {
                var iframeWindow;
                var iframeDocument;
                var target;

                if (!iframe || !iframe.contentWindow) {
                    return;
                }

                try {
                    iframeWindow = iframe.contentWindow;
                    iframeDocument = iframeWindow.document;

                    /*
                    * Tracciamento reload iframe.
                    * Se Complianz provoca un reload interno, mostriamo stato di salvataggio.
                    */
                    try {
                        iframeWindow.addEventListener('beforeunload', function () {
                            iframeIsLoading = true;

                            if (hasConsentInteraction) {
                                showSaving();
                            }
                        });

                        iframeWindow.addEventListener('pagehide', function () {
                            iframeIsLoading = true;

                            if (hasConsentInteraction) {
                                showSaving();
                            }
                        });
                    } catch (error) {
                        // noop.
                    }

                    /*
                    * Tracciamento fetch dentro iframe.
                    * Serve per non chiudere mentre Complianz sta salvando via fetch().
                    */
                    try {
                        if (
                            iframeWindow.fetch &&
                            !iframeWindow.__imgCmplzTcfPopupFetchPatched
                        ) {
                            iframeWindow.__imgCmplzTcfPopupFetchPatched = true;

                            var originalFetch = iframeWindow.fetch;

                            iframeWindow.fetch = function () {
                                pendingAjaxRequests++;

                                return originalFetch.apply(this, arguments)
                                    .then(function (response) {
                                        return response;
                                    })
                                    .catch(function (error) {
                                        throw error;
                                    })
                                    .finally(function () {
                                        pendingAjaxRequests = Math.max(0, pendingAjaxRequests - 1);
                                    });
                            };
                        }
                    } catch (error) {
                        // noop.
                    }

                    /*
                    * Tracciamento XMLHttpRequest dentro iframe.
                    * Serve se Complianz usa admin-ajax.php o REST via XHR classico.
                    */
                    try {
                        if (
                            iframeWindow.XMLHttpRequest &&
                            !iframeWindow.__imgCmplzTcfPopupXhrPatched
                        ) {
                            iframeWindow.__imgCmplzTcfPopupXhrPatched = true;

                            var OriginalXHR = iframeWindow.XMLHttpRequest;

                            iframeWindow.XMLHttpRequest = function () {
                                var xhr = new OriginalXHR();
                                var originalSend = xhr.send;

                                xhr.send = function () {
                                    pendingAjaxRequests++;

                                    xhr.addEventListener('loadend', function () {
                                        pendingAjaxRequests = Math.max(0, pendingAjaxRequests - 1);
                                    });

                                    return originalSend.apply(xhr, arguments);
                                };

                                return xhr;
                            };
                        }
                    } catch (error) {
                        // noop.
                    }

                    if (!iframeDocument || !iframeDocument.documentElement) {
                        return;
                    }

                    target = iframeDocument.getElementById(config.targetHash);

                    if (target) {
                        window.setTimeout(function () {
                            try {
                                target.scrollIntoView({
                                    behavior: 'auto',
                                    block: 'start',
                                    inline: 'nearest'
                                });
                            } catch (error) {
                                target.scrollIntoView(true);
                            }
                        }, 100);

                        window.setTimeout(function () {
                            try {
                                target.scrollIntoView({
                                    behavior: 'auto',
                                    block: 'start',
                                    inline: 'nearest'
                                });
                            } catch (error) {
                                target.scrollIntoView(true);
                            }
                        }, 600);
                    }

                    /*
                    * Tracciamo le interazioni dell'utente con checkbox, switch, label,
                    * bottoni e controlli dentro la sezione Complianz.
                    */
                    iframeDocument.addEventListener('change', function (event) {
                        if (!event.target || !event.target.closest) {
                            return;
                        }

                        if (
                            event.target.matches('input, select, textarea') ||
                            event.target.closest('#' + config.targetHash) ||
                            event.target.closest('.cmplz-document') ||
                            event.target.closest('.cmplz-category') ||
                            event.target.closest('.cmplz-service') ||
                            event.target.closest('.cmplz-vendor')
                        ) {
                            markConsentInteraction();
                        }
                    }, true);

                    iframeDocument.addEventListener('click', function (event) {
                        var interactiveElement;

                        if (!event.target || !event.target.closest) {
                            return;
                        }

                        interactiveElement = event.target.closest(
                            'input, label, button, select, textarea, .cmplz-btn, .cmplz-category, .cmplz-service, .cmplz-vendor'
                        );

                        if (!interactiveElement) {
                            return;
                        }

                        if (
                            interactiveElement.closest('#' + config.targetHash) ||
                            interactiveElement.closest('.cmplz-document') ||
                            interactiveElement.closest('.cmplz-category') ||
                            interactiveElement.closest('.cmplz-service') ||
                            interactiveElement.closest('.cmplz-vendor')
                        ) {
                            markConsentInteraction();
                        }
                    }, true);

                    iframeDocument.addEventListener('click', function (event) {
                        var actionButton;

                        if (!event.target || !event.target.closest) {
                            return;
                        }

                        actionButton = event.target.closest(config.consentActionSelector);

                        if (!actionButton) {
                            return;
                        }

                        /*
                         * Lasciamo prima lavorare Complianz dentro iframe.
                         * Poi ricarichiamo la pagina padre per sincronizzare banner,
                         * stato consenso, cookie e script bloccati/sbloccati.
                         */
                        window.setTimeout(reloadParentAfterConsent, 700);
                    }, true);

                    iframeDocument.addEventListener('submit', function () {
                        window.setTimeout(reloadParentAfterConsent, 700);
                    }, true);
                } catch (error) {
                    /*
                     * Se l'iframe non fosse accessibile, il contenuto rimane comunque
                     * visibile e la navigazione interna continua a funzionare.
                     */
                }
            }

            function prepareTrigger(trigger) {
                var hrefAttribute;

                if (!trigger) {
                    return;
                }

                trigger.setAttribute('data-img-cmplz-tcf-popup-ready', 'true');
                trigger.setAttribute('aria-haspopup', 'dialog');
                trigger.setAttribute('aria-controls', config.modalId);

                /*
                 * Aggiorniamo data-img-cmplz-tcf-popup-original-href solo se href
                 * è davvero una URL valida. In questo modo correggiamo anche
                 * eventuali valori legacy già presenti come "#".
                 */
                hrefAttribute = trigger.getAttribute('href') || '';

                if (isPotentiallyValidRawUrl(hrefAttribute)) {
                    trigger.setAttribute('data-img-cmplz-tcf-popup-original-href', hrefAttribute);
                }
            }

            function prepareTriggers() {
                var triggers = document.querySelectorAll(config.triggerSelector);

                Array.prototype.forEach.call(triggers, prepareTrigger);
            }

            function handleTriggerClick(event) {
                var trigger = closest(event.target, config.triggerSelector);

                if (!trigger) {
                    return;
                }

                stopEvent(event);
                prepareTrigger(trigger);
                openModal(trigger);
            }

            function handleKeydown(event) {
                if (
                    event.key === 'Escape' &&
                    modal &&
                    modal.classList.contains(config.visibleClass)
                ) {
                    event.preventDefault();
                    requestCloseModal();
                }
            }

            function startObserver() {
                if (observer || typeof window.MutationObserver !== 'function') {
                    return;
                }

                observer = new MutationObserver(function () {
                    prepareTriggers();
                });

                observer.observe(document.documentElement, {
                    childList: true,
                    subtree: true
                });
            }

            /*
             * Capture phase:
             * intercettiamo prima del comportamento standard Complianz/browser.
             */
            document.addEventListener('click', handleTriggerClick, true);
            document.addEventListener('keydown', handleKeydown, true);

            document.addEventListener('DOMContentLoaded', function () {
                prepareTriggers();
                startObserver();
            });

            window.addEventListener('load', function () {
                prepareTriggers();
                startObserver();
            });

            prepareTriggers();
            startObserver();

            window.setTimeout(prepareTriggers, 300);
            window.setTimeout(prepareTriggers, 1000);
            window.setTimeout(prepareTriggers, 2000);
        })();
    </script>
